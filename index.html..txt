// Mot de passe (majuscules et espaces ignorés)
var PASSWORD = "B2P_Forever";

function $(id){return document.getElementById(id)}
function norm(s){return s.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g,"").replace(/\s+/g,"")}
function tryOpen(){
  if(norm($("pw").value)===norm(PASSWORD)){
    $("lock").classList.add("hide");$("inside").classList.remove("hide");window.scrollTo(0,0);
  }else{
    $("fb").textContent="Ce n’est pas le bon mot de passe, réessaie ma princesse.";
    var c=$("lock");c.classList.remove("shake");void c.offsetWidth;c.classList.add("shake");
  }
}
function tab(n){
  $("letter").classList.toggle("hide",n!==1);$("song").classList.toggle("hide",n!==2);
  $("t1").classList.toggle("on",n===1);$("t2").classList.toggle("on",n===2);
}
$("go").onclick=tryOpen;
$("pw").addEventListener("keydown",function(e){if(e.key==="Enter"){tryOpen()}});
$("t1").onclick=function(){tab(1)};
$("t2").onclick=function(){tab(2)};
